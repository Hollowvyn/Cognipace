  rows:intervals.map((b,i)=>{const success=[5,10,0,0,1,3,0,2][i];return {...b,completedReviews:counts[i],goodEasy:success,validRatings:counts[i],reviewSuccess:counts[i]?success/counts[i]:null,evidence:counts[i]?'measured':'not-measured'}}),

 const memoryRows=state==='edges'?[...emptyPeriods.map(b=>({...b,...emptyMemory})),...memory.rows.map((row,i)=>i===memory.rows.length-1?{...row,...emptyMemory}:row)]:selectRows(memory.rows)
 const displayMemory={...memory,rows:memoryRows,scale:buildAdaptiveDurationScale(memoryRows.flatMap(row=>[row.medianStrengthDays,row.q1,row.q3].filter((n):n is number=>n!==null)))}

  <div className="fixture-shell"><aside className="fixture-sidebar"><b>CogniPace</b><p>Analytics</p></aside><div className="fixture-content"><div className="cp-analytics-history grid gap-4">
   <AnalyticsChartPanel {...chartProps} id="recall" title="Observed Recall vs FSRS Estimate" question="How did recalled review outcomes compare with the FSRS estimate?" description="Rating-derived recalled outcomes compared with reconstructed FSRS retrievability immediately before those exact reviews."><ObservedRecallVsFsrsView view={displayRecall} timeFrame={timeFrame}/></AnalyticsChartPanel>
   <div className="cp-analytics-history-pair grid min-w-0 gap-4">
    <AnalyticsChartPanel {...chartProps} id="memory" title="Memory Strength" question="Are your reviewed memories staying strong for longer?" description="FSRS’s reconstructed post-review estimate of how long the memories reviewed in each bucket may remain retrievable."><MemoryStrengthView view={displayMemory} timeFrame={timeFrame}/></AnalyticsChartPanel>
    <AnalyticsChartPanel {...chartProps} id="practice" title="Practice Rhythm" question="When you practiced more or less, how did Review Success move?" description="Completed review volume and the Good + Easy share move together by time bucket; the relationship is association only."><PracticeRhythmView view={displayPractice} timeFrame={timeFrame}/>{state==='mixed'?<div className="mt-3"><AnalyticsReadinessState compact title="Practice Rhythm" recommendedRange={null} readiness={{ready:false,requestedDays:30,bucketDays:3,requestedBuckets:10,effectiveBuckets:8,effectiveStart:'2026-09-09',assessments:12,minimumAssessments:24,activeBuckets:6,minimumActiveBuckets:6,longestGap:1,maximumGap:2,gapRuns:2,maximumGapRuns:2,failingReasons:['insufficient-assessments']}}/></div>:null}</AnalyticsChartPanel>
   </div>
   <div className="grid w-full max-w-[64rem] gap-4 lg:grid-cols-2"><AnalyticsChartPanel {...chartProps} id="ratings" title="Ratings Mix" question="How is the balance of your review ratings changing?" description="The changing share of valid Again, Hard, Good, and Easy review ratings across the selected period."><RatingsMixView view={displayRatings} timeFrame={timeFrame}/></AnalyticsChartPanel><div/></div>
  </div></div></div>
 </SurfaceRoot>
}
