import { useMemo, useState } from 'react'
import { createRoot } from 'react-dom/client'
import './proof.css'
import { SurfaceRoot } from '@/components/ui/surface'
import { AnalyticsChartPanel } from '@/features/analytics/components/analytics-chart-panel'
import { AnalyticsTargetEditor } from '@/features/analytics/components/analytics-target-editor'
import { NewProblemSuccessView } from '@/features/analytics/components/new-problem-success-view'
import { ObservedRecallVsFsrsView } from '@/features/analytics/components/recall-ratings-views'
import { buildHistoricalAnalyticsViews, applyHistoricalChartTargets } from '@/features/analytics/domain/historical-presentation'
import { buildAnalyticsTimeFrame } from '@/features/analytics/domain/analytics-time'
import { buildAnalyticsBucketsFromTimeFrame } from '@/features/analytics/domain/analytics-range-policy'
import { normalizeFsrsSchedulingOptions } from '@/lib/fsrs'
import { defaultAnalyticsTargets, type AnalyticsTargets } from '@/features/settings/domain'
import type { AnalyticsReviewEvent } from '@/features/analytics/domain/chart-data'

const ratings = ['again','hard','good','easy'] as const
const compositions = [[4,4,2,0],[3,4,3,0],[0,0,0,0],[2,3,5,0],[3,2,5,0],[2,2,6,0],[0,0,0,0],[2,4,4,0],[2,2,6,0],[2,0,8,0]]
const repeats = [12,15,0,9,10,16,8,0,15,10]

function history(scenario:string, timeFrame:ReturnType<typeof buildAnalyticsTimeFrame>) {
  const buckets=buildAnalyticsBucketsFromTimeFrame(timeFrame)
  const events:AnalyticsReviewEvent[]=[]
  const event=(id:string,cardId:string,rating:string,reviewedAt:Date):AnalyticsReviewEvent=>({id,cardId,problemSlug:cardId,title:cardId,topicLabels:[],rating,reviewedAt,isCorrect:null,fsrsReviewLog:null})
  if(scenario==='Empty')return events
  if(scenario!=='First attempts only') for(let i=0;i<8;i++)events.push(event(`initial-${i}`,`repeat-card-${i}`,'good',new Date(buckets[0]!.start.getTime()-30*86400000+i*3600000)))
  buckets.forEach((bucket,index)=>{
    const time=new Date(Math.min(timeFrame.asOf ? new Date(timeFrame.asOf).getTime()-3600000 : Infinity, (bucket.start.getTime()+bucket.end.getTime())/2))
    const active=scenario==='Singleton'?index===5:scenario==='Sparse'?[2,5,8].includes(index):true
    if(!active)return
    if(scenario!=='Repeat reviews only') {
      const counts=scenario==='Sparse'?(index===2?[2,0,0,0]:index===5?[1,1,0,0]:[0,1,1,0]):scenario==='Singleton'?[2,1,2,0]:compositions[index]!
      counts.forEach((count,ratingIndex)=>{for(let j=0;j<count;j++){const id=`new-${index}-${ratingIndex}-${j}`;events.push(event(id,id,ratings[ratingIndex]!,new Date(time.getTime()+j*1000)))}})
      if(index===2&&scenario==='Mixed practice')events.push(event('invalid-first','invalid-first','unknown',time))
    }
    if(scenario!=='First attempts only') {
      const n=scenario==='Sparse'?3:scenario==='Singleton'?5:repeats[index]!
      for(let j=0;j<n;j++)events.push(event(`repeat-${index}-${j}`,`repeat-card-${j%8}`,scenario==='Sparse'&&index===2?'again':j%5===0?'again':j%4===0?'hard':'good',new Date(time.getTime()+j*5000)))
    }
  })
  return events
}

function App() {
  const [theme,setTheme]=useState<'dark'|'light'>('dark')
  const [scenario,setScenario]=useState('Mixed practice')
  const [targets,setTargets]=useState<AnalyticsTargets>({...defaultAnalyticsTargets,targetFirstAttemptSuccess:.8,targetFirstAttemptGoodEasy:.6})
  const [failSave,setFailSave]=useState(false)
  const [saving,setSaving]=useState(false)
  const timeFrame=useMemo(()=>buildAnalyticsTimeFrame({asOf:new Date(scenario==='Cross year'?'2026-01-11T06:44:00.000Z':'2026-10-03T06:44:00.000Z'),requestedDays:30,timeZone:'America/New_York'}),[scenario])
  const views=useMemo(()=>{
    const buckets=buildAnalyticsBucketsFromTimeFrame(timeFrame)
    return buildHistoricalAnalyticsViews(history(scenario,timeFrame),{buckets,start:buckets[0]!.start,end:new Date(timeFrame.asOf),fsrsOptions:normalizeFsrsSchedulingOptions({targetRetention:.87}),timeZone:timeFrame.timeZone,timeFrame,analyticsTargets:defaultAnalyticsTargets})
  },[scenario,timeFrame])
  const fitted=applyHistoricalChartTargets(views,targets)
  const save=async(patch:Partial<AnalyticsTargets>)=>{
    setSaving(true)
    try {
      await new Promise(resolve=>setTimeout(resolve,200))
      if(failSave)throw new Error('Fixture save failed. Try again.')
      const next={...targets,...patch};setTargets(next);return next
    } finally {setSaving(false)}
  }
  return <SurfaceRoot surface="dashboard" theme={theme} className="min-h-screen">
    <nav className="proof-controls border-b border-border text-xs" aria-label="Fixture controls">
      <span>Production components · illustrative history · local fixture saves</span>
      <button type="button" onClick={()=>setTheme(theme==='dark'?'light':'dark')}>Switch to {theme==='dark'?'light':'dark'}</button>
      <label>Scenario <select aria-label="Fixture scenario" value={scenario} onChange={e=>setScenario(e.target.value)}>{['Mixed practice','Sparse','Singleton','First attempts only','Repeat reviews only','Cross year','Empty'].map(s=><option key={s}>{s}</option>)}</select></label>
      <label><input type="checkbox" checked={failSave} onChange={e=>setFailSave(e.target.checked)} /> Fail fixture save</label>
      <span role="status">{saving?'Saving fixture goal…':''}</span>
    </nav>
    <main className="proof-content cp-analytics-history">
      <div className="grid min-w-0 gap-4 lg:grid-cols-2">
        <AnalyticsChartPanel historical id="new-problem-success" title="New Problem Success" question="How are your first recorded outcomes changing?" description="First recorded assessment per problem. Success = Hard + Good + Easy.">
          <NewProblemSuccessView view={fitted.firstAttemptOutcomes} timeFrame={timeFrame} targetControl={<div className="cp-first-attempt-targets flex flex-wrap gap-x-4 gap-y-2"><AnalyticsTargetEditor metric="firstAttemptSuccess" targets={targets} onSave={save}/><AnalyticsTargetEditor metric="firstAttemptGoodEasy" targets={targets} onSave={save}/></div>}/>
        </AnalyticsChartPanel>
        <AnalyticsChartPanel historical id="observed-recall-vs-fsrs" title="Recall vs FSRS Estimate" question="How well are you recalling problems on later reviews?" description="Repeat assessments only. Both curves use the same paired reviews.">
          <ObservedRecallVsFsrsView view={fitted.observedRecallVsFsrs} timeFrame={timeFrame} targetControl={<AnalyticsTargetEditor metric="recall" targets={targets} onSave={save}/>} />
        </AnalyticsChartPanel>
      </div>
    </main>
  </SurfaceRoot>
}
createRoot(document.getElementById('root')!).render(<App/> )
